def _update_read_offsets_impl(
    read_offsets: torch.Tensor,
    groups: list[torch.Tensor],
    next_tokens: torch.Tensor,
    num_reqs: torch.Tensor,
) -> None:
    """Post-verify read-offset scatter for every mamba group in one program.
    Each of `groups` is a [width] slot tensor; `next_tokens` the rejection
    sampler output ([>= num_reqs, K+1], INVALID_TOKEN_ID-padded); `num_reqs`
    a 0-d tensor, so it enters as a value, not a shape. Rows past it take
    offset 0 (their slots are the null block)."""
    width = groups[0].shape[0]
    rows = next_tokens.shape[0]
    if rows >= width:
        next_tokens = next_tokens[:width]
    else:
        next_tokens = torch.nn.functional.pad(
            next_tokens, (0, 0, 0, width - rows), value=INVALID_TOKEN_ID
        )
    num_valid = (next_tokens != INVALID_TOKEN_ID).sum(dim=1)
    valid = torch.arange(width, device=read_offsets.device) < num_reqs
    offsets = torch.where(valid, (num_valid - 1).clamp(min=0), 0).to(
        read_offsets.dtype
    )
    indices = torch.cat(groups, dim=0).long() if len(groups) > 1 else groups[0].long()
    read_offsets.index_put_((indices,), offsets.repeat(len(groups)))


def _verify_fused_impl(
    logits: torch.Tensor,
    bonus_logits_indices: torch.Tensor,
    target_logits_indices: torch.Tensor,
    temperatures: torch.Tensor,
    top_k: torch.Tensor,
    top_p: torch.Tensor,
    bonus_u: torch.Tensor,
    accept_u: torch.Tensor,
    recover_u: torch.Tensor,
    draft_token_ids: torch.Tensor,
    draft_lengths: torch.Tensor,
    segment_ids: torch.Tensor,
    group_indices: torch.Tensor,
    synthetic_conditional_rates: torch.Tensor | None,
    max_draft_tokens: int,
    synthetic_mode: bool,
    topk_candidates: int,
    num_reqs: torch.Tensor | None = None,
):
    """Non-greedy spec verify of one chunk as one program
    (VLLM_TPU_VERIFY_FUSED): the bonus/target logit gathers, the bonus-token
    sample, the per-position sampling-parameter gathers and the rejection
    sampler, op for op the same as `_sample_spec_verify_chunk`'s separate
    programs. The uniforms come from the caller's eager draws, in the same
    order and shapes, so the random stream is unchanged."""
    bonus_logits = torch.index_select(logits, 0, bonus_logits_indices)
    target_logits = torch.index_select(logits, 0, target_logits_indices)
    bonus_token_ids = _sample_from_logits_impl(
        bonus_logits, temperatures, bonus_u, top_k, top_p, topk_candidates
    ).view(-1)
    next_tokens = _random_rejection_sample_impl(
        draft_token_ids,
        target_logits,
        draft_lengths,
        bonus_token_ids,
        segment_ids,
        group_indices,
        max_draft_tokens,
        temperatures[segment_ids],
        top_k[segment_ids],
        top_p[segment_ids],
        accept_u,
        recover_u,
        synthetic_conditional_rates,
        synthetic_mode,
        topk_candidates,
    )
    if num_reqs is None:
        return next_tokens
    # Per-row read offsets (`_update_read_offsets_impl` before its group
    # pad/scatter): rows past num_reqs take 0.
    num_valid = (next_tokens != INVALID_TOKEN_ID).sum(dim=1)
    valid = torch.arange(next_tokens.shape[0], device=next_tokens.device) < num_reqs
    row_offsets = torch.where(valid, (num_valid - 1).clamp(min=0), 0)
    return next_tokens, row_offsets

def _update_mamba_slot_read_offsets(
    self,
    indices_per_group: list[torch.Tensor] | None,
    next_tokens: torch.Tensor | None,
    num_reqs: int,
) -> None:
    """Scatter this chunk's mamba read offsets into the slot-indexed buffer.

    Each entry holds one mamba group's base slot per batch position for
    the chunk (padded tail = slot 0, the null block, so its writes are
    harmless); all groups get the same offsets, since the buffer is
    indexed by slot/block id and groups allocate distinct ones. The
    compact pool shares one slot tensor across groups and so passes a
    single-element list.

    The offset is `num_accepted - 1` derived from the rejection-sampler
    output for verify chunks, and 0 for non-spec chunks (prefill / plain
    decode) — which also resets a group's offset after prefill. The next
    step's GDN kernel resumes each request from checkpoint `offset` of
    its state block (the checkpoint of its last accepted token).
    """
    if self.mamba_slot_read_offsets is None or not indices_per_group:
        return
    if (
        next_tokens is not None
        and envs.VLLM_TPU_FUSED_READ_OFFSET_UPDATE
        and self.mamba_slot_read_offsets.device.type == "tpu"
    ):
        _update_read_offsets_compiled(
            self.mamba_slot_read_offsets,
            list(indices_per_group),
            next_tokens,
            self._num_reqs_tensor(num_reqs),
        )
        return
    offsets = torch.zeros(
        indices_per_group[0].shape[0],
        dtype=torch.int32,
        device=indices_per_group[0].device,
    )
    if next_tokens is not None:
        # num_valid = accepted drafts + 1 (bonus); the checkpoint of the
        # last accepted token is at offset num_valid - 1.
        num_valid = (
            (next_tokens[:num_reqs] != INVALID_TOKEN_ID).sum(dim=1).to(torch.int32)
        )
        offsets[:num_reqs] = (num_valid - 1).clamp(min=0)
    if len(indices_per_group) == 1:
        self.mamba_slot_read_offsets.index_put_(
            (indices_per_group[0].long(),), offsets
        )
    else:
        all_indices = torch.cat(indices_per_group, dim=0).long()
        all_offsets = offsets.repeat(len(indices_per_group))
        self.mamba_slot_read_offsets.index_put_((all_indices,), all_offsets)


def _sample_spec_verify_fused(
    self,
    logits: torch.Tensor,
    md: SpecDecodeMetadata,
    cur_start_idx: int,
    cur_end_idx: int,
    sampling_generator: torch.Generator | None,
) -> torch.Tensor:
    """`_sample_spec_verify_chunk`'s non-greedy branch as one program
    (`_verify_fused_impl`). The three uniform draws stay eager, with the
    shapes, dtype and order of the separate path, so the random stream
    and the selected tokens are the same."""
    assert md.draft_token_ids is not None
    assert self.rejection_sampler is not None
    assert self.speculative_config is not None
    padded_num_reqs = md.bonus_logits_indices.shape[0]
    vocab = logits.shape[-1]
    temps, top_k, top_p = self._padded_sampling_params(
        cur_start_idx, cur_end_idx, padded_num_reqs, logits.dtype, logits.device
    )
    topk_candidates = self._sampler_topk_candidates(cur_start_idx, cur_end_idx, None)
    host_phases.mark("vfy_params")
    bonus_u = torch.rand(
        (padded_num_reqs, vocab),
        dtype=torch.float32,
        device=logits.device,
        generator=sampling_generator,
    )
    accept_u = torch.rand(
        md.draft_token_ids.shape,
        dtype=torch.float32,
        device=logits.device,
        generator=sampling_generator,
    )
    recover_u = torch.rand(
        (md.target_logits_indices.shape[0], vocab),
        dtype=torch.float32,
        device=logits.device,
        generator=sampling_generator,
    )
    note = getattr(self, "_eager_shape_note", None)
    if note is not None:
        note(
            "fused_draws",
            (tuple(bonus_u.shape), tuple(accept_u.shape), tuple(recover_u.shape)),
        )
    host_phases.mark("vfy_rand")
    args = (
        logits,
        md.bonus_logits_indices,
        md.target_logits_indices,
        temps,
        top_k,
        top_p,
        bonus_u,
        accept_u,
        recover_u,
        md.draft_token_ids,
        md.draft_lengths,
        md.segment_ids,
        md.group_indices,
        self.rejection_sampler.synthetic_conditional_rates,
        self.speculative_config.num_speculative_tokens,
        self.rejection_sampler.synthetic_mode,
        topk_candidates,
    )
    ro_groups, ro_num_reqs = self._fused_read_offsets
    fold_ro = (
        envs.VLLM_TPU_VERIFY_FUSED_READ_OFFSETS
        and self.mamba_slot_read_offsets is not None
        and bool(ro_groups)
        and envs.VLLM_TPU_FUSED_READ_OFFSET_UPDATE
        and self.mamba_slot_read_offsets.device.type == "tpu"
    )
    if fold_ro:
        out, row_offsets = _verify_fused_compiled(
            *args, self._num_reqs_tensor(ro_num_reqs)
        )
        # Pad/truncate to the slot groups' width and scatter eagerly, instead
        # of a compiled program that mutates the slot buffer (and its aot
        # mutation epilogue). Padded rows' slots are the null block.
        groups = list(ro_groups)
        width = groups[0].shape[0]
        rows = row_offsets.shape[0]
        vals = (
            row_offsets[:width]
            if rows >= width
            else torch.nn.functional.pad(row_offsets, (0, width - rows))
        ).to(self.mamba_slot_read_offsets.dtype)
        if len(groups) > 1:
            idx = torch.cat(groups, dim=0).long()
            vals = vals.repeat(len(groups))
        else:
            idx = groups[0].long()
        ref_buf = (
            self.mamba_slot_read_offsets.clone()
            if self._verify_fused_checks_left > 0
            else None
        )
        self.mamba_slot_read_offsets.index_put_((idx,), vals)
        self._fused_read_offsets_applied = True
        if ref_buf is not None:
            # Check: the separate program's scatter on a copy must match.
            _update_read_offsets_impl(
                ref_buf, groups, out, self._num_reqs_tensor(ro_num_reqs)
            )
            ro_mism = (ref_buf != self.mamba_slot_read_offsets).sum()
            self._verify_fused_ro_mismatches = (
                ro_mism
                if self._verify_fused_ro_mismatches is None
                else self._verify_fused_ro_mismatches + ro_mism
            )
    else:
        out = _verify_fused_compiled(*args)
    if self._verify_fused_checks_left > 0:
        self._check_verify_fused(out, args)
    return out


def _sample_spec_verify_vps(
    self,
    local_logits: torch.Tensor,
    md: SpecDecodeMetadata,
    cur_start_idx: int,
    cur_end_idx: int,
    sampling_generator: torch.Generator | None,
    fallback_rows: int = 0,
) -> torch.Tensor:
    """`_sample_spec_verify_fused` on the rank's vocab shard (VLLM_TPU_VPS_VERIFY).

    The three uniform draws are the fused path's, in the same order and at
    the full vocabulary width, so the random stream is unchanged; the read
    offsets take the separate update in sample_tokens."""
    vh = self._vps_head
    assert vh is not None and md.draft_token_ids is not None
    assert self.rejection_sampler is not None
    assert self.speculative_config is not None
    padded_num_reqs = md.bonus_logits_indices.shape[0]
    vocab = vh.full_vocab
    if self._vps_cand_draws:
        # PD DEC-FB-2 VPB: one uniform per candidate column (TP x KC), no
        # full-vocabulary draws (VLLM_TPU_VPS_DRAWS=candidate).
        from vllm_torchtpu.layers.adapter.sample import vp_sampler

        vocab = vh.tp_size * vp_sampler.VP_KC
    temps, top_k, top_p = self._padded_sampling_params(
        cur_start_idx, cur_end_idx, padded_num_reqs, local_logits.dtype, local_logits.device
    )
    host_phases.mark("vfy_params")
    bonus_u = torch.rand(
        (padded_num_reqs, vocab),
        dtype=torch.float32,
        device=local_logits.device,
        generator=sampling_generator,
    )
    accept_u = torch.rand(
        md.draft_token_ids.shape,
        dtype=torch.float32,
        device=local_logits.device,
        generator=sampling_generator,
    )
    recover_u = torch.rand(
        (md.target_logits_indices.shape[0], vocab),
        dtype=torch.float32,
        device=local_logits.device,
        generator=sampling_generator,
    )
    note = getattr(self, "_eager_shape_note", None)
    if note is not None:
        note(
            "vps_draws",
            (tuple(bonus_u.shape), tuple(accept_u.shape), tuple(recover_u.shape)),
        )
    host_phases.mark("vfy_rand")
    args = (
        md.bonus_logits_indices,
        md.target_logits_indices,
        temps,
        top_k,
        top_p,
        bonus_u,
        accept_u,
        recover_u,
        md.draft_token_ids,
        md.draft_lengths,
        md.segment_ids,
        md.group_indices,
        self.rejection_sampler.synthetic_conditional_rates,
        self.speculative_config.num_speculative_tokens,
        self.rejection_sampler.synthetic_mode,
    )
    if self._vps_fb_path:
        out = self._vps_verify_fb(
            local_logits, args, cur_end_idx - cur_start_idx, fallback_rows, sampling_generator
        )
    else:
        out = _vps_verify_compiled(local_logits, vh.shard_offset, *args)
    if self._vps_checks_left > 0:
        self._check_vps(out, local_logits, args, cur_start_idx, cur_end_idx)
    return out


def _vps_verify_fb(
    self,
    local_logits: torch.Tensor,
    args: tuple,
    num_active: int,
    fallback_rows: int,
    sampling_generator: torch.Generator | None,
) -> torch.Tensor:
    """PD DEC-FB-2: `_vps_verify_fb_compiled` on the chunk (args as
    `_vps_verify_compiled`'s). Under per-candidate draws the F fallback rows
    draw their own full-vocabulary uniforms, after the three stock draws.
    The [flagged, covered, uncovered] row counts accumulate on device and
    are logged as VPS_FALLBACK every VLLM_TPU_VPS_STATS_EVERY chunks."""
    vh = self._vps_head
    assert vh is not None
    fb_u = None
    if self._vps_cand_draws and fallback_rows > 0:
        fb_u = torch.rand(
            (fallback_rows, vh.full_vocab),
            dtype=torch.float32,
            device=local_logits.device,
            generator=sampling_generator,
        )
        note = getattr(self, "_eager_shape_note", None)
        if note is not None:
            note("vps_fb_u", tuple(fb_u.shape))
    n_active = self._upload_unless_unchanged(
        "vps_num_active", torch.tensor([num_active], dtype=torch.int32)
    )
    out, stats = _vps_verify_fb_compiled(
        local_logits,
        vh.shard_offset,
        *args,
        n_active,
        fallback_rows,
        self._vps_cand_draws,
        fb_u,
    )
    stats = stats.to(torch.int64)
    self._vps_fb_stats = stats if self._vps_fb_stats is None else self._vps_fb_stats + stats
    self._vps_fb_chunks += 1
    every = vps_wiring.stats_every()
    if every and self._vps_fb_chunks % every == 0:
        flagged, covered, uncovered = self._vps_fb_stats.tolist()
        logger.info(
            "VPS_FALLBACK chunks=%d flagged_rows=%d covered_rows=%d "
            "uncovered_rows=%d max_rows=%d tiegap_slots=%d draws=%s",
            self._vps_fb_chunks,
            flagged,
            covered,
            uncovered,
            self._vps_fb_max,
            self._vps_tiegap,
            "candidate" if self._vps_cand_draws else "full",
        )
    return out


def _precompile_vps_verify(
    self,
    dummy_logits: torch.Tensor,
    bonus_logits_indices: torch.Tensor,
    target_logits_indices: torch.Tensor,
    draft_token_ids: torch.Tensor,
    draft_lengths: torch.Tensor,
    segment_ids: torch.Tensor,
    group_indices: torch.Tensor,
    warm_accept_u: torch.Tensor,
    served_only: bool = False,
) -> None:
    """VLLM_TPU_VPS_VERIFY (PD DEC-FB-2 warm-up): the VPS verify program at
    one (padded_logits_length, padded_num_reqs) bucket, at the dtypes
    `_sample_spec_verify_vps` passes: today's `_vps_verify_compiled`, or
    with the fallback / draw flags `_vps_verify_fb_compiled` at every
    fallback row count the chunk dispatch can pick."""
    from vllm_torchtpu.layers.adapter.sample import vp_sampler

    vh = self._vps_head
    assert vh is not None
    assert self.rejection_sampler is not None
    assert self.speculative_config is not None
    num_reqs = bonus_logits_indices.shape[0]
    width = vh.tp_size * vp_sampler.VP_KC if self._vps_cand_draws else vh.full_vocab
    local = torch.zeros(
        (dummy_logits.shape[0], vh.shard_width), dtype=dummy_logits.dtype
    ).to(self.device)
    temps = torch.ones((num_reqs, 1), dtype=dummy_logits.dtype).to(self.device)
    top_k = torch.zeros((num_reqs, 1), dtype=torch.int32).to(self.device)
    top_p = torch.ones((num_reqs, 1), dtype=torch.float32).to(self.device)
    bonus_u = torch.zeros((num_reqs, width), dtype=torch.float32).to(self.device)
    recover_u = torch.zeros(
        (target_logits_indices.shape[0], width), dtype=torch.float32
    ).to(self.device)
    args = (
        bonus_logits_indices,
        target_logits_indices,
        temps,
        top_k,
        top_p,
        bonus_u,
        warm_accept_u,
        recover_u,
        draft_token_ids,
        draft_lengths,
        segment_ids,
        group_indices,
        self.rejection_sampler.synthetic_conditional_rates,
        self.speculative_config.num_speculative_tokens,
        self.rejection_sampler.synthetic_mode,
    )
    if not self._vps_fb_path:
        synchronize_tensors(_vps_verify_compiled(local, vh.shard_offset, *args))
        if getattr(envs, "VLLM_TPU_VPS_WARM_DRAWS", 0):
            self._warm_vps_draws(
                num_reqs, draft_token_ids, target_logits_indices, width, ()
            )
        return
    n_active = torch.tensor([num_reqs], dtype=torch.int32).to(self.device)
    variants = vps_wiring.fallback_variants(self._vps_fb_max, self._vps_tiegap)
    if served_only:
        # The width a step whose every request has an in-range top_k gets
        # (need 0, as `_vps_fallback_rows` computes it); others compile lazily.
        served = vps_wiring.fallback_slots(
            0, self._vps_tiegap, max(self._vps_fb_max, self._vps_tiegap)
        )
        variants = [v for v in variants if v == served]
    for rows in variants:
        fb_u = None
        if self._vps_cand_draws and rows > 0:
            fb_u = torch.zeros((rows, vh.full_vocab), dtype=torch.float32).to(
                self.device
            )
        out, stats = _vps_verify_fb_compiled(
            local,
            vh.shard_offset,
            *args,
            n_active,
            rows,
            self._vps_cand_draws,
            fb_u,
        )
        synchronize_tensors(out)
        synchronize_tensors(stats)
        if getattr(envs, "VLLM_TPU_VPS_WARM_DRAWS", 0):
            self._warm_vps_draws(
                num_reqs,
                draft_token_ids,
                target_logits_indices,
                width,
                ((rows, vh.full_vocab),) if self._vps_cand_draws and rows > 0 else (),
            )


def _warm_vps_draws(
    self,
    num_reqs: int,
    draft_token_ids: torch.Tensor,
    target_logits_indices: torch.Tensor,
    width: int,
    fb_shapes: tuple,
) -> None:
    """q38 L4 EAGER-FIX E-b (VLLM_TPU_VPS_WARM_DRAWS=N): issue a served step's
    eager uniform draws at this bucket N times, in the serving order (bonus,
    accept, recover, then fb_u), with the shapes, dtype and device
    `_sample_spec_verify_vps` / `_vps_verify_fb` use, so torch_tpu compiles
